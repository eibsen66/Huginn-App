# configuring a target firmware project, generating objects or linking.

function(idf_build_get_property result property)
    get_property(value GLOBAL PROPERTY ${property})
    set(${result} "${value}" PARENT_SCOPE)
endfunction()
function(idf_build_set_property property value)
    if("APPEND" IN_LIST ARGN)
        set_property(GLOBAL APPEND PROPERTY ${property} "${value}")
    else()
        set_property(GLOBAL PROPERTY ${property} "${value}")
    endif()
endfunction()
function(__kconfig_component_init target)
    # Kconfig evaluation is outside this resolver-only probe.
endfunction()
